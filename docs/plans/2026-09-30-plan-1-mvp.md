> 📘 **觀念：** **ORM**（SQLAlchemy）讓我們用 Python 類別描述資料表，一個類別對應一張表，一個屬性對應一個欄位。資料庫會負責守住規則：`ForeignKey`（外鍵）保證持股一定對應到存在的股票，`CheckConstraint`（檢查條件）保證股數大於 0。就算程式有 bug，壞資料也寫不進去。

```python
from pathlib import Path

from sqlalchemy import MetaData, create_engine, event
from sqlalchemy.engine import Engine, make_url
from sqlalchemy.orm import DeclarativeBase, sessionmaker

from app.config import settings

# 固定約束的命名規則，Alembic 之後修改或刪除約束時才找得到名字
NAMING_CONVENTION = {
    "ix": "ix_%(column_0_label)s",
    "uq": "uq_%(table_name)s_%(column_0_name)s",
    "ck": "ck_%(table_name)s_%(constraint_name)s",
    "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s",
    "pk": "pk_%(table_name)s",
}


class Base(DeclarativeBase):
    metadata = MetaData(naming_convention=NAMING_CONVENTION)


def make_engine(url: str, **kwargs) -> Engine:
    is_sqlite = url.startswith("sqlite")
    if is_sqlite:
        kwargs.setdefault("connect_args", {"check_same_thread": False})
        database = make_url(url).database
        if database and database != ":memory:":
            Path(database).parent.mkdir(parents=True, exist_ok=True)

    engine = create_engine(url, **kwargs)

    if is_sqlite:
        @event.listens_for(engine, "connect")
        def _sqlite_pragmas(dbapi_conn, _record):
            cursor = dbapi_conn.cursor()
            cursor.execute("PRAGMA foreign_keys=ON")   # SQLite 預設不檢查外鍵
            cursor.execute("PRAGMA journal_mode=WAL")  # 爬蟲寫入時 API 仍可讀取
            cursor.close()

    return engine


engine = make_engine(settings.database_url)
SessionLocal = sessionmaker(bind=engine, expire_on_commit=False)
```

- [ ] **Step 5：實作** `app/models.py`

```python
import datetime as dt
from decimal import Decimal

from sqlalchemy import CheckConstraint, Date, DateTime, ForeignKey, Numeric, String, Text
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.db import Base

ROLES = ("legislator", "councilor")
PARSE_STATUSES = ("pending", "parsed", "verified", "failed")
OWNERS = ("本人", "配偶", "未成年子女")
MARKETS = ("TWSE", "TPEx", "ESB")
CRAWL_STATUSES = ("running", "success", "failed")


def _in(column: str, values: tuple[str, ...]) -> str:
    return f"{column} IN ({', '.join(repr(v) for v in values)})"


class Politician(Base):
    __tablename__ = "politician"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(50))
    name_alt: Mapped[str | None] = mapped_column(String(100))
    photo_url: Mapped[str | None] = mapped_column(Text)

    terms: Mapped[list["Term"]] = relationship(
        back_populates="politician", order_by="Term.start_date", cascade="all, delete-orphan")
    declarations: Mapped[list["Declaration"]] = relationship(
        back_populates="politician", cascade="all, delete-orphan")

    @property
    def current_term(self) -> "Term | None":
        """最新（開始日期最晚）的任期。"""
        return self.terms[-1] if self.terms else None


class Term(Base):
    __tablename__ = "term"
    __table_args__ = (CheckConstraint(_in("role", ROLES), name="role"),)

    id: Mapped[int] = mapped_column(primary_key=True)
    politician_id: Mapped[int] = mapped_column(
        ForeignKey("politician.id", ondelete="CASCADE"), index=True)
    role: Mapped[str] = mapped_column(String(20))
    session: Mapped[str] = mapped_column(String(20))
    district: Mapped[str | None] = mapped_column(String(50))
    party: Mapped[str] = mapped_column(String(50))
    start_date: Mapped[dt.date] = mapped_column(Date)
    end_date: Mapped[dt.date | None] = mapped_column(Date)

    politician: Mapped[Politician] = relationship(back_populates="terms")


class Declaration(Base):
    __tablename__ = "declaration"
    __table_args__ = (CheckConstraint(_in("parse_status", PARSE_STATUSES), name="parse_status"),)

    id: Mapped[int] = mapped_column(primary_key=True)
    politician_id: Mapped[int] = mapped_column(
        ForeignKey("politician.id", ondelete="CASCADE"), index=True)
    declared_date: Mapped[dt.date] = mapped_column(Date)
    source_url: Mapped[str] = mapped_column(Text)
    pdf_sha256: Mapped[str | None] = mapped_column(String(64), unique=True)
    parse_status: Mapped[str] = mapped_column(String(10), default="pending")

    politician: Mapped[Politician] = relationship(back_populates="declarations")
    holdings: Mapped[list["Holding"]] = relationship(
        back_populates="declaration", cascade="all, delete-orphan")


class Stock(Base):
    __tablename__ = "stock"
    __table_args__ = (CheckConstraint(_in("market", MARKETS), name="market"),)

    code: Mapped[str] = mapped_column(String(10), primary_key=True)
    name: Mapped[str] = mapped_column(String(100))
    market: Mapped[str] = mapped_column(String(4))


class Holding(Base):
    __tablename__ = "holding"
    __table_args__ = (
        CheckConstraint("shares > 0", name="shares_positive"),
        CheckConstraint(_in("owner", OWNERS), name="owner"),
    )

    id: Mapped[int] = mapped_column(primary_key=True)
    declaration_id: Mapped[int] = mapped_column(
        ForeignKey("declaration.id", ondelete="CASCADE"), index=True)
    stock_code: Mapped[str] = mapped_column(ForeignKey("stock.code"), index=True)
    owner: Mapped[str] = mapped_column(String(10))
    shares: Mapped[int]
    raw_text: Mapped[str | None] = mapped_column(Text)

    declaration: Mapped[Declaration] = relationship(back_populates="holdings")
    stock: Mapped[Stock] = relationship()


class StockPrice(Base):
    __tablename__ = "stock_price"

    stock_code: Mapped[str] = mapped_column(ForeignKey("stock.code"), primary_key=True)
    date: Mapped[dt.date] = mapped_column(Date, primary_key=True)
    close: Mapped[Decimal] = mapped_column(Numeric(12, 2))


class CrawlLog(Base):
    __tablename__ = "crawl_log"
    __table_args__ = (CheckConstraint(_in("status", CRAWL_STATUSES), name="status"),)

    id: Mapped[int] = mapped_column(primary_key=True)
    source: Mapped[str] = mapped_column(String(50))
    started_at: Mapped[dt.datetime] = mapped_column(DateTime)
    finished_at: Mapped[dt.datetime | None] = mapped_column(DateTime)
    status: Mapped[str] = mapped_column(String(10))
    rows: Mapped[int] = mapped_column(default=0)
    message: Mapped[str | None] = mapped_column(Text)
```

> 📘 `Mapped[str]` 代表 NOT NULL（必填），`Mapped[str | None]` 代表可以是空值。`datetime` 模組刻意用 `dt` 別名匯入，因為 `StockPrice` 有一個欄位就叫 `date`，名稱會打架。

預期：產生 `alembic.ini` 和 `migrations/` 資料夾。`alembic.ini` 裡的 `sqlalchemy.url` 不用改，連線位址會由 `env.py` 從 `settings` 讀取。

- [ ] **Step 2：整份取代** `migrations/env.py`

```python
from logging.config import fileConfig

from alembic import context

import app.models  # noqa: F401  註冊所有資料表
from app.config import settings
from app.db import Base, make_engine

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    context.configure(
        url=settings.database_url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        render_as_batch=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    engine = make_engine(settings.database_url)
    with engine.connect() as connection:
        # render_as_batch：SQLite 不支援大部分 ALTER TABLE，改用「重建表」的方式修改
        context.configure(connection=connection, target_metadata=target_metadata,
                          render_as_batch=True)
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
import datetime as dt
from decimal import Decimal, InvalidOperation


def parse_roc_date(text: str) -> dt.date:
    """'1150929' → date(2026, 9, 29)。格式不對就丟出 ValueError。"""
    text = text.strip()
    if not text.isdigit() or len(text) not in (6, 7):
        raise ValueError(f"不是民國日期：{text!r}")
    year = int(text[:-4]) + 1911
    return dt.date(year, int(text[-4:-2]), int(text[-2:]))


def parse_price(text: str | None) -> Decimal | None:
    """價格字串轉 Decimal；沒成交（空白、--、---、0）一律回傳 None，絕不回傳 0。"""
    if text is None:
        return None
    cleaned = text.strip().replace(",", "")
    try:
        value = Decimal(cleaned)
    except InvalidOperation:
        return None
    if not value.is_finite() or value <= 0:
        return None
    return value
```

> - **重試 + 指數退避**：網路偶爾會斷，遇到逾時或 5xx 就等 1 秒、2 秒、4 秒再試；遇到 404 這種「再試也沒用」的錯誤就直接放棄。

```python
import datetime as dt
import time
from collections.abc import Callable
from typing import Any

import httpx
from sqlalchemy.orm import Session, sessionmaker

from app.models import CrawlLog

USER_AGENT = "legislator-stocks-class-project/0.1"

# 爬蟲工作：接收 session，回傳（寫入筆數, 備註訊息）
Job = Callable[[Session], tuple[int, str | None]]


class CrawlError(Exception):
    """資料格式不符合預期這類錯誤，整批資料都不應該寫入。"""


def fetch_json(url: str, *, params: dict | None = None, retries: int = 3,
               backoff: float = 1.0, client: httpx.Client | None = None,
               sleep: Callable[[float], None] = time.sleep) -> Any:
    own_client = client is None
    if client is None:
        client = httpx.Client(timeout=30, follow_redirects=True,
                              headers={"User-Agent": USER_AGENT})
    try:
        for attempt in range(retries + 1):
            try:
                response = client.get(url, params=params)
                response.raise_for_status()
                return response.json()
            except (httpx.TransportError, httpx.HTTPStatusError) as exc:
                retryable = (isinstance(exc, httpx.TransportError)
                             or exc.response.status_code >= 500)
                if not retryable or attempt == retries:
                    raise
                sleep(backoff * 2 ** attempt)
    finally:
        if own_client:
            client.close()


def _utcnow() -> dt.datetime:
    return dt.datetime.now(dt.timezone.utc).replace(tzinfo=None)


def run_crawler(session_factory: sessionmaker, source: str, job: Job) -> CrawlLog:
    """執行一個爬蟲工作，並在 crawl_log 留下紀錄。工作的寫入與成功紀錄在同一個交易中。"""
    with session_factory() as session:
        log = CrawlLog(source=source, started_at=_utcnow(), status="running", rows=0)
        session.add(log)
        session.commit()

        try:
            rows, message = job(session)
            log.status, log.rows, log.message = "success", rows, message
        except Exception as exc:  # noqa: BLE001  任何錯誤都要記錄下來，不能讓 cron 默默失敗
            session.rollback()
            log.status, log.message = "failed", f"{type(exc).__name__}: {exc}"

        log.finished_at = _utcnow()
        session.commit()
        session.refresh(log)
        return log
```

```python
import datetime as dt
from dataclasses import dataclass
from decimal import Decimal
from typing import Any

from sqlalchemy.orm import Session

from app.crawler.base import CrawlError
from app.crawler.utils import parse_price, parse_roc_date
from app.models import Stock, StockPrice


@dataclass(frozen=True)
class PriceRow:
    code: str
    name: str
    market: str
    date: dt.date
    close: Decimal


def parse_quotes(data: Any, *, market: str, code_key: str, name_key: str,
                 close_key: str, date_key: str = "Date") -> tuple[list[PriceRow], int]:
    """把證交所／櫃買中心的行情 JSON 轉成 PriceRow。回傳（資料列, 跳過筆數）。"""
    if not isinstance(data, list) or not data or not isinstance(data[0], dict):
        raise CrawlError("回應是空的，或不是物件清單")
    missing = {date_key, code_key, name_key, close_key} - set(data[0])
    if missing:
        raise CrawlError(f"回應缺少欄位 {sorted(missing)}，API 格式可能改了")

    rows: dict[str, PriceRow] = {}
    skipped = 0
    for item in data:
        code = (item.get(code_key) or "").strip()
        close = parse_price(item.get(close_key))
        if not code or close is None:
            skipped += 1
            continue
        try:
            day = parse_roc_date(item.get(date_key) or "")
        except ValueError:
            skipped += 1
            continue
        rows[code] = PriceRow(code, (item.get(name_key) or "").strip(), market, day, close)
    return list(rows.values()), skipped


def _dialect_insert(session: Session):
    """SQLite 與 PostgreSQL 的 upsert 寫法相同，只是要從不同模組匯入。"""
    if session.get_bind().dialect.name == "postgresql":
        from sqlalchemy.dialects.postgresql import insert
    else:
        from sqlalchemy.dialects.sqlite import insert
    return insert


def save_prices(session: Session, rows: list[PriceRow]) -> int:
    if not rows:
        return 0
    insert = _dialect_insert(session)

    stock_stmt = insert(Stock)
    stock_stmt = stock_stmt.on_conflict_do_update(
        index_elements=["code"],
        set_={"name": stock_stmt.excluded["name"], "market": stock_stmt.excluded["market"]},
    )
    session.execute(stock_stmt, [{"code": r.code, "name": r.name, "market": r.market}
                                 for r in rows])

    price_stmt = insert(StockPrice)
    price_stmt = price_stmt.on_conflict_do_update(
        index_elements=["stock_code", "date"],
        set_={"close": price_stmt.excluded["close"]},
    )
    session.execute(price_stmt, [{"stock_code": r.code, "date": r.date, "close": r.close}
                                 for r in rows])
    return len(rows)


def skipped_message(skipped: int) -> str | None:
    return f"跳過 {skipped} 筆沒有收盤價的資料" if skipped else None
```

```python
import csv
import datetime as dt
from pathlib import Path

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models import OWNERS, ROLES, Declaration, Holding, Politician, Stock, Term

POLITICIAN_COLUMNS = ["id", "name", "name_alt", "photo_url", "role", "session",
                      "district", "party", "start_date", "end_date"]
HOLDING_COLUMNS = ["politician_id", "declared_date", "source_url", "stock_code",
                   "owner", "shares"]


class CsvImportError(ValueError):
    pass


def _read(path: Path, columns: list[str]) -> list[tuple[int, dict[str, str]]]:
    # utf-8-sig：Excel 存的 CSV 開頭會有 BOM，這個編碼會自動去掉
    with open(path, encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f)
        missing = set(columns) - set(reader.fieldnames or [])
        if missing:
            raise CsvImportError(f"{path} 缺少欄位：{sorted(missing)}")
        return list(enumerate(reader, start=2))  # 第 1 行是標題


def _optional(value: str | None) -> str | None:
    value = (value or "").strip()
    return value or None


def _required(row: dict[str, str], field: str, line: int) -> str:
    value = _optional(row.get(field))
    if value is None:
        raise CsvImportError(f"第 {line} 行：{field} 不能是空的")
    return value


def _int(row: dict[str, str], field: str, line: int) -> int:
    text = _required(row, field, line).replace(",", "")
    try:
        return int(text)
    except ValueError:
        raise CsvImportError(f"第 {line} 行：{field} 不是整數：{text!r}") from None


def _date(row: dict[str, str], field: str, line: int) -> dt.date:
    text = _required(row, field, line)
    try:
        return dt.date.fromisoformat(text)
    except ValueError:
        raise CsvImportError(f"第 {line} 行：{field} 不是 YYYY-MM-DD 格式：{text!r}") from None


def import_politicians(session: Session, path: Path) -> int:
    rows = _read(path, POLITICIAN_COLUMNS)
    for line, row in rows:
        politician_id = _int(row, "id", line)
        role = _required(row, "role", line)
        if role not in ROLES:
            raise CsvImportError(f"第 {line} 行：role 只能是 {ROLES}，收到 {role!r}")
        term_session = _required(row, "session", line)

        politician = session.get(Politician, politician_id) or Politician(id=politician_id)
        politician.name = _required(row, "name", line)
        politician.name_alt = _optional(row["name_alt"])
        politician.photo_url = _optional(row["photo_url"])
        session.add(politician)

        term = next((t for t in politician.terms
                     if t.role == role and t.session == term_session), None)
        if term is None:
            term = Term(role=role, session=term_session)
            politician.terms.append(term)
        term.district = _optional(row["district"])
        term.party = _required(row, "party", line)
        term.start_date = _date(row, "start_date", line)
        term.end_date = _date(row, "end_date", line) if _optional(row["end_date"]) else None

    session.flush()
    return len(rows)


def import_holdings(session: Session, path: Path) -> int:
    rows = _read(path, HOLDING_COLUMNS)
    known_stocks = set(session.scalars(select(Stock.code)))
    groups: dict[tuple[int, dt.date], dict] = {}

    for line, row in rows:
        politician_id = _int(row, "politician_id", line)
        if session.get(Politician, politician_id) is None:
            raise CsvImportError(
                f"第 {line} 行：找不到 politician_id={politician_id}，請先匯入 politicians CSV")
        code = _required(row, "stock_code", line)
        if code not in known_stocks:
            raise CsvImportError(
                f"第 {line} 行：股票代號 {code} 不在資料庫，請先執行 python -m app.crawler prices")
        owner = _required(row, "owner", line)
        if owner not in OWNERS:
            raise CsvImportError(f"第 {line} 行：owner 只能是 {OWNERS}，收到 {owner!r}")
        shares = _int(row, "shares", line)
        if shares <= 0:
            raise CsvImportError(f"第 {line} 行：shares 必須大於 0")

        key = (politician_id, _date(row, "declared_date", line))
        group = groups.setdefault(key, {"source_url": _required(row, "source_url", line),
                                        "holdings": []})
        group["holdings"].append(Holding(stock_code=code, owner=owner, shares=shares))

    for (politician_id, declared_date), group in groups.items():
        declaration = session.scalars(select(Declaration).where(
            Declaration.politician_id == politician_id,
            Declaration.declared_date == declared_date,
        )).first()
        if declaration is None:
            declaration = Declaration(politician_id=politician_id, declared_date=declared_date)
            session.add(declaration)
        declaration.source_url = group["source_url"]
        declaration.parse_status = "verified"
        declaration.holdings = group["holdings"]  # delete-orphan 會刪掉舊的持股

    session.flush()
    return sum(len(g["holdings"]) for g in groups.values())
```

```python
import argparse
import sys
from pathlib import Path

from app.crawler import esb_prices, tpex_prices, twse_prices
from app.crawler.base import Job, run_crawler
from app.crawler.declarations.csv_import import import_holdings, import_politicians
from app.db import SessionLocal

PRICE_JOBS: dict[str, Job] = {
    "twse_prices": twse_prices.job,
    "tpex_prices": tpex_prices.job,
    "esb_prices": esb_prices.job,
}


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="python -m app.crawler")
    sub = parser.add_subparsers(dest="command", required=True)
    sub.add_parser("prices", help="抓上市＋上櫃＋興櫃股價")
    for name in PRICE_JOBS:
        sub.add_parser(name)
    for name in ("import_politicians", "import_holdings"):
        sub.add_parser(name).add_argument("csv", type=Path)
    args = parser.parse_args(argv)

    if args.command == "prices":
        jobs = dict(PRICE_JOBS)
    elif args.command in PRICE_JOBS:
        jobs = {args.command: PRICE_JOBS[args.command]}
    elif args.command == "import_politicians":
        jobs = {args.command: lambda s: (import_politicians(s, args.csv), str(args.csv))}
    else:
        jobs = {args.command: lambda s: (import_holdings(s, args.csv), str(args.csv))}

    ok = True
    for name, job in jobs.items():
        log = run_crawler(SessionLocal, name, job)
        print(f"[{log.status}] {name}: {log.rows} 筆 {log.message or ''}")
        ok = ok and log.status == "success"
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main())
```

```python
from dataclasses import dataclass, field

    stock_code: str
    stock_name: str
    owner: str
    shares: int
    close: Decimal | None
    market_value: Decimal | None


@dataclass
class Portfolio:
    declaration: Declaration | None
    holdings: list[HoldingValue] = field(default_factory=list)
    price_date: dt.date | None = None

    @property
    def market_value(self) -> Decimal | None:
        if self.declaration is None:
            return None
        return sum((h.market_value for h in self.holdings if h.market_value is not None),
                   Decimal(0))

    @property
    def missing_price(self) -> bool:
        return any(h.close is None for h in self.holdings)


def latest_declarations(session: Session,
                        politician_ids: Iterable[int] | None = None) -> dict[int, Declaration]:
    """每位立委最新一份可用的申報，key 是 politician_id。"""
    stmt = (
        select(Declaration)
        .where(Declaration.parse_status.in_(USABLE_STATUSES))
        .options(selectinload(Declaration.holdings).selectinload(Holding.stock))
        .order_by(Declaration.politician_id, Declaration.declared_date.desc(),
                  Declaration.id.desc())
    )
    if politician_ids is not None:
        stmt = stmt.where(Declaration.politician_id.in_(list(politician_ids)))
    result: dict[int, Declaration] = {}
    for declaration in session.scalars(stmt):
        result.setdefault(declaration.politician_id, declaration)  # 排序後第一份就是最新的
    return result


def latest_prices(session: Session, codes: Iterable[str]) -> dict[str, StockPrice]:
    """每檔股票最新一天的收盤價。沒有任何股價的股票不會出現在結果中。"""
    codes = list(codes)
    if not codes:
        return {}
    newest = (
        select(StockPrice.stock_code, func.max(StockPrice.date).label("max_date"))
        .where(StockPrice.stock_code.in_(codes))
        .group_by(StockPrice.stock_code)
        .subquery()
    )
    stmt = select(StockPrice).join(newest, and_(
        StockPrice.stock_code == newest.c.stock_code,
        StockPrice.date == newest.c.max_date,
    ))
    return {price.stock_code: price for price in session.scalars(stmt)}


def build_portfolio(declaration: Declaration | None,
                    prices: dict[str, StockPrice]) -> Portfolio:
    if declaration is None:
        return Portfolio(declaration=None)
    holdings: list[HoldingValue] = []
    price_dates: list[dt.date] = []
    for h in declaration.holdings:
        price = prices.get(h.stock_code)
        close = price.close if price else None
        if price:
            price_dates.append(price.date)
        holdings.append(HoldingValue(
            stock_code=h.stock_code, stock_name=h.stock.name, owner=h.owner, shares=h.shares,
            close=close, market_value=close * h.shares if close is not None else None,
        ))
    holdings.sort(key=lambda v: v.market_value or Decimal(0), reverse=True)
    return Portfolio(declaration, holdings, max(price_dates) if price_dates else None)


def portfolios(session: Session, politician_ids: list[int]) -> dict[int, Portfolio]:
    declarations = latest_declarations(session, politician_ids)
    codes = {h.stock_code for d in declarations.values() for h in d.holdings}
    prices = latest_prices(session, codes)
    return {pid: build_portfolio(declarations.get(pid), prices) for pid in politician_ids}
```

```python
def register_error_handlers(app: FastAPI) -> None:
    @app.exception_handler(APIError)
    async def _api_error(request: Request, exc: APIError):
        return _error(exc.status_code, exc.code, exc.message)

    @app.exception_handler(StarletteHTTPException)
    async def _http_error(request: Request, exc: StarletteHTTPException):
        code = {404: "not_found", 405: "method_not_allowed"}.get(exc.status_code, "http_error")
        return _error(exc.status_code, code, str(exc.detail))

    @app.exception_handler(RequestValidationError)
    async def _validation_error(request: Request, exc: RequestValidationError):
        return _error(422, "validation_error", "參數格式錯誤",
                      details=jsonable_encoder(exc.errors()))

    @app.exception_handler(Exception)
    async def _unexpected(request: Request, exc: Exception):
        return _error(500, "internal_error", "伺服器發生錯誤")
```

```python
from fastapi import APIRouter, Query

def _summary(p: Politician, pf: Portfolio) -> dict:
    term = p.current_term
    return {
        "id": p.id, "name": p.name, "name_alt": p.name_alt, "photo_url": p.photo_url,
        "party": term.party if term else None,
        "session": term.session if term else None,
        "declared_date": pf.declaration.declared_date if pf.declaration else None,
        "market_value": to_float(pf.market_value),
        "missing_price": pf.missing_price,
    }


def _sort(items: list[dict], sort: str) -> list[dict]:
    """依欄位排序；該欄位是 null 的一律排在最後（不論升冪降冪）。"""
    key = sort.lstrip("-")
    present = [i for i in items if i[key] is not None]
    absent = [i for i in items if i[key] is None]
    present.sort(key=lambda i: i[key], reverse=sort.startswith("-"))
    return present + absent


@router.get("", response_model=PoliticianPage)
def list_politicians(
    session: SessionDep,
    party: str | None = None,
    sort: PoliticianSort = "-market_value",
    limit: int = Query(20, ge=1, le=100),
    page: int = Query(1, ge=1),
):
    politicians = session.scalars(
        select(Politician).options(selectinload(Politician.terms)).order_by(Politician.id)
    ).all()
    if party is not None:
        politicians = [p for p in politicians
                       if p.current_term and p.current_term.party == party]

    pfs = portfolios(session, [p.id for p in politicians])
    items = _sort([_summary(p, pfs[p.id]) for p in politicians], sort)
    start = (page - 1) * limit
    return {"items": items[start:start + limit], "total": len(items),
            "page": page, "limit": limit}


@router.get("/{politician_id}", response_model=PoliticianDetail)

    politician = session.get(Politician, politician_id)
    if politician is None:
        raise APIError(404, "not_found", "找不到這位立委")

    pf = portfolios(session, [politician.id])[politician.id]
    term = politician.current_term
    return {
        **_summary(politician, pf),
        "district": term.district if term else None,
        "price_date": pf.price_date,
        "holdings": [
```

```python
from collections import defaultdict
from typing import Literal

from fastapi import APIRouter, Query
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.api.deps import SessionDep
from app.api.errors import APIError
from app.api.schemas import StockDetail, StockPage, to_float

from app.services.valuation import latest_declarations, latest_prices

router = APIRouter(prefix="/stocks", tags=["stocks"])

StockSort = Literal["-holders", "holders", "code"]


def _holdings_by_stock(session: Session) -> dict[str, list[tuple[Declaration, Holding]]]:
    """以每位立委最新可用的申報為準，依股票代號分組。"""
    result: dict[str, list[tuple[Declaration, Holding]]] = defaultdict(list)
    for declaration in latest_declarations(session).values():
        for holding in declaration.holdings:
            result[holding.stock_code].append((declaration, holding))
    return result


@router.get("", response_model=StockPage)
def list_stocks(
    session: SessionDep,
    sort: StockSort = "-holders",
    limit: int = Query(20, ge=1, le=100),
    page: int = Query(1, ge=1),
):
    by_stock = _holdings_by_stock(session)
    prices = latest_prices(session, by_stock)
    stocks = session.scalars(select(Stock).where(Stock.code.in_(list(by_stock)))).all()

    items = []
    for stock in stocks:
        price = prices.get(stock.code)
        items.append({
            "code": stock.code, "name": stock.name, "market": stock.market,
            "holders": len({d.politician_id for d, _ in by_stock[stock.code]}),
            "close": to_float(price.close) if price else None,
            "price_date": price.date if price else None,
        })
    items.sort(key=lambda i: i["code"])
    if sort != "code":  # Python 的排序是穩定的，人數相同時會保留代號順序
        items.sort(key=lambda i: i["holders"], reverse=sort == "-holders")

    start = (page - 1) * limit
    return {"items": items[start:start + limit], "total": len(items),
            "page": page, "limit": limit}


@router.get("/{code}", response_model=StockDetail)
def get_stock(code: str, session: SessionDep):
    stock = session.get(Stock, code)
    if stock is None:
        raise APIError(404, "not_found", "找不到這檔股票")

    price = latest_prices(session, [code]).get(code)
    holders = []
    for declaration, holding in _holdings_by_stock(session).get(code, []):
        politician = declaration.politician
        term = politician.current_term
        holders.append({
            "politician_id": politician.id, "name": politician.name,
            "party": term.party if term else None,
            "owner": holding.owner, "shares": holding.shares,
            "market_value": to_float(price.close * holding.shares) if price else None,
        })
    holders.sort(key=lambda h: h["shares"], reverse=True)

    return {"code": stock.code, "name": stock.name, "market": stock.market,
            "close": to_float(price.close) if price else None,
            "price_date": price.date if price else None, "holders": holders}
```

```python
import datetime as dt
import sqlite3
import sys
from contextlib import closing
from pathlib import Path

from sqlalchemy.engine import make_url

from app.config import settings


def backup(db_path: Path, backup_dir: Path, keep: int = 7,
           today: dt.date | None = None) -> Path:
    backup_dir.mkdir(parents=True, exist_ok=True)
    target = backup_dir / f"app-{(today or dt.date.today()).isoformat()}.db"
    with closing(sqlite3.connect(db_path)) as src, closing(sqlite3.connect(target)) as dst:
        src.backup(dst)
    for old in sorted(backup_dir.glob("app-*.db"))[:-keep]:
        old.unlink()
    return target


def main() -> int:
    url = make_url(settings.database_url)
    if not url.drivername.startswith("sqlite") or not url.database:
        print("只支援 SQLite 檔案資料庫的備份")
        return 1
    db_path = Path(url.database)
    print(f"備份完成：{backup(db_path, db_path.parent / 'backups')}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
